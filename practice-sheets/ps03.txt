#Q19
#!/bin/bash
target_dir="$1"
if [ ! -d "$target_dir" ]; then
    echo "Error: Directory does not exist"
    exit 1
fi
count=0
for file in "$target_dir"/*.bak
do
    if [ -f "$file" ]; then
        echo "Found backup: $file"
        count=$((count + 1))
    fi
done
if [ $count -eq 0 ]; then
    echo "No backup files found." >> backup_log.txt
    echo "No backups found. Logged to backup_log.txt"
else
    echo "Total backups found: $count"
fi
exit 0

#!/bin/bash
file="$1"
if [ $# -ne 1 ]; then
    echo "Usage: $0 <file>"
    exit 1
fi
if [ ! -f "$file" ]; then
    echo "File does not exist"
    exit 1
fi
if [ ! -r "$file" ]; then
    echo "File is not readable"
    exit 1
fi
read lines words chars <<< "$(wc -l -w -c < "$file")"
echo "File: $file"
echo "Lines: $lines"
echo "Words: $words"
echo "Characters: $chars"
echo "$(date) - $0 processed $file" >> execution_history.log
exit 0
